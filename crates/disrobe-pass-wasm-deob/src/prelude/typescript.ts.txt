const v128Lanes = (v: bigint, bits: bigint): bigint[] => {
  const out: bigint[] = [];
  const count: number = Number(128n / bits);
  const mask: bigint = (1n << bits) - 1n;
  for (let i = 0; i < count; i++) out.push((v >> (BigInt(i) * bits)) & mask);

};
const v128FromLanes = (lanes: bigint[], bits: bigint): bigint => {

  return out & V128_MASK;
};
