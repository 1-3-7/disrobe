static inline void wasm_atomic_waiter_unlock(void) { pthread_mutex_unlock(&wasm_atomic_waiter_mutex); }

static inline int32_t wasm_atomic_wait_at(size_t p, int wide_value, int64_t expected, int64_t timeout_ns) {
  wasm_atomic_waiter_lock();
  int64_t current = wide_value ? (int64_t)wasm_atomic_u64_load(p) : (int64_t)(int32_t)wasm_atomic_u32_load(p);
  if (current != expected) { wasm_atomic_waiter_unlock(); return 1; }
  if (timeout_ns == 0) { wasm_atomic_waiter_unlock(); return 2; }
  int slot = -1;
  for (int index = 0; index < WASM_ATOMIC_WAITER_CAP; index++) { if (!wasm_atomic_waiters[index].live) { slot = index; break; } }
  if (slot < 0) { wasm_atomic_waiter_unlock(); wasm_atomic_trap("atomic-waiter-capacity"); }
  wasm_atomic_waiters[slot].address = (uint64_t)p;
  wasm_atomic_waiters[slot].live = 1;
  wasm_atomic_waiters[slot].notified = 0;
  uint64_t started = wasm_atomic_now_ns();
  for (;;) {
    if (wasm_atomic_waiters[slot].notified) { wasm_atomic_waiters[slot].live = 0; wasm_atomic_waiter_unlock(); return 0; }
    if (timeout_ns < 0) { wasm_atomic_waiter_park(-1); continue; }
    uint64_t elapsed = wasm_atomic_now_ns() - started;
    if (elapsed >= (uint64_t)timeout_ns) { wasm_atomic_waiters[slot].live = 0; wasm_atomic_waiter_unlock(); return 2; }
    wasm_atomic_waiter_park((int64_t)((uint64_t)timeout_ns - elapsed));
  }
}
static inline int32_t wasm_atomic_notify_at(size_t p, int32_t count) {
  uint32_t limit = (uint32_t)count;
  uint32_t woken = 0;
  wasm_atomic_waiter_lock();
  for (int index = 0; index < WASM_ATOMIC_WAITER_CAP && woken < limit; index++) {
    if (wasm_atomic_waiters[index].live && !wasm_atomic_waiters[index].notified && wasm_atomic_waiters[index].address == (uint64_t)p) { wasm_atomic_waiters[index].notified = 1; woken++; }
  }
  wasm_atomic_waiter_unlock();
  if (woken > 0) wasm_atomic_waiter_wake();
  return (int32_t)woken;
}
