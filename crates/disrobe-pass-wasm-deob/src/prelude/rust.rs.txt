    static WASM_GLOBALS: RefCell<Vec<i64>> = RefCell::new(vec![0i64; 64]);
}

#[inline]
fn wasm_mem_addr(addr: i32, offset: u64) -> usize {
    (addr as u32 as u64).wrapping_add(offset) as usize
}

fn wasm_load_i32(addr: i32, offset: u64) -> i32 {
    WASM_MEMORY.with(|m| {
        let m = m.borrow();
        let a = wasm_mem_addr(addr, offset);
        let mut b = [0u8; 4];
        if a + 4 <= m.len() { b.copy_from_slice(&m[a..a + 4]); }
        i32::from_le_bytes(b)
    })
}
fn wasm_load_i64(addr: i32, offset: u64) -> i64 {
    WASM_MEMORY.with(|m| {
        let m = m.borrow();
        let a = wasm_mem_addr(addr, offset);
        let mut b = [0u8; 8];
        if a + 8 <= m.len() { b.copy_from_slice(&m[a..a + 8]); }
        i64::from_le_bytes(b)
    })
}
fn wasm_load_f32(addr: i32, offset: u64) -> f32 { f32::from_bits(wasm_load_i32(addr, offset) as u32) }
fn wasm_load_f64(addr: i32, offset: u64) -> f64 { f64::from_bits(wasm_load_i64(addr, offset) as u64) }
fn wasm_load_i32_8u(addr: i32, offset: u64) -> i32 {
    WASM_MEMORY.with(|m| { let m = m.borrow(); let a = wasm_mem_addr(addr, offset); *m.get(a).unwrap_or(&0) as i32 })
}
fn wasm_load_i32_8s(addr: i32, offset: u64) -> i32 { (wasm_load_i32_8u(addr, offset) as i8) as i32 }
fn wasm_load_i32_16u(addr: i32, offset: u64) -> i32 {
    WASM_MEMORY.with(|m| { let m = m.borrow(); let a = wasm_mem_addr(addr, offset); let mut b = [0u8; 2]; if a + 2 <= m.len() { b.copy_from_slice(&m[a..a + 2]); } u16::from_le_bytes(b) as i32 })
}

fn wasm_load_i64_8u(addr: i32, offset: u64) -> i64 { wasm_load_i32_8u(addr, offset) as u32 as i64 }

fn wasm_load_i64_16u(addr: i32, offset: u64) -> i64 { wasm_load_i32_16u(addr, offset) as u32 as i64 }
fn wasm_load_i64_16s(addr: i32, offset: u64) -> i64 { (wasm_load_i32_16u(addr, offset) as i16) as i64 }
fn wasm_load_i64_32u(addr: i32, offset: u64) -> i64 { wasm_load_i32(addr, offset) as u32 as i64 }
fn wasm_load_i64_32s(addr: i32, offset: u64) -> i64 { wasm_load_i32(addr, offset) as i64 }

fn wasm_store_i32(addr: i32, offset: u64, val: i32) {
    WASM_MEMORY.with(|m| { let mut m = m.borrow_mut(); let a = wasm_mem_addr(addr, offset); if a + 4 <= m.len() { m[a..a + 4].copy_from_slice(&val.to_le_bytes()); } });
}
fn wasm_store_i64(addr: i32, offset: u64, val: i64) {
    WASM_MEMORY.with(|m| { let mut m = m.borrow_mut(); let a = wasm_mem_addr(addr, offset); if a + 8 <= m.len() { m[a..a + 8].copy_from_slice(&val.to_le_bytes()); } });
}

}

fn wasm_global_get(index: u32) -> i64 {
    WASM_GLOBALS.with(|g| g.borrow().get(index as usize).copied().unwrap_or(0))
}
fn wasm_global_set(index: u32, val: i64) {
    WASM_GLOBALS.with(|g| { let mut g = g.borrow_mut(); if let Some(slot) = g.get_mut(index as usize) { *slot = val; } });
}
fn wasm_memory_size() -> i32 {
    WASM_MEMORY.with(|m| (m.borrow().len() / (64 * 1024)) as i32)
}
fn wasm_memory_grow(delta: i32) -> i32 {
    WASM_MEMORY.with(|m| {
        let mut m = m.borrow_mut();
        let prev = (m.len() / (64 * 1024)) as i32;
        if delta > 0 {
            let new_len = m.len() + (delta as usize) * 64 * 1024;
            m.resize(new_len, 0u8);
        }
        prev
    })
}
