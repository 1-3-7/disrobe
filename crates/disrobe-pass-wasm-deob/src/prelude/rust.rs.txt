#![allow(dead_code, unused_variables, unused_mut, unused_assignments, unused_labels, unused_parens, unreachable_code, clippy::all)]
//! disrobe wasm lift target=rust
//! Self-contained WebAssembly numeric + memory runtime for the lifted functions below.
//! Divide/remainder by zero and signed-overflow are made total (return 0) to keep the
//! artifact panic-free; float->int truncation saturates (Rust `as`), matching trunc_sat.
//! Single-threaded runtime: memory.atomic.wait reports 1 (not-equal) or 2 (timed-out)
//! and never blocks, memory.atomic.notify wakes nobody, and atomic accesses share the
//! same storage as their plain counterparts.

use std::cell::RefCell;

thread_local! {
    static WASM_MEMORY: RefCell<Vec<u8>> = RefCell::new(vec![0u8; 64 * 1024]);
    static WASM_GLOBALS: RefCell<Vec<i64>> = RefCell::new(vec![0i64; 64]);
}

#[inline]
fn wasm_mem_addr(addr: i32, offset: u64) -> usize {
    (addr as u32 as u64).wrapping_add(offset) as usize
}

fn wasm_atomic_trap(reason: &str) -> ! {
    eprintln!("DR-WASMDEOB-TRAP/1:{reason}");
    panic!("wasm atomic trap");
}

fn wasm_atomic_range(addr: u64, offset: u64, width: u64) {
    if width == 0 || !width.is_power_of_two() {
        wasm_atomic_trap("atomic-invalid-width");
    }
    let aligned_start: u64 = addr.wrapping_add(offset);
    if aligned_start % width != 0 {
        wasm_atomic_trap("atomic-unaligned");
    }
    let start: u64 = match addr.checked_add(offset) {
        Some(value) => value,
        None => wasm_atomic_trap("atomic-oob"),
    };
    let end: u64 = match start.checked_add(width) {
        Some(value) => value,
        None => wasm_atomic_trap("atomic-oob"),
    };
    let memory_len: u64 = WASM_MEMORY.with(|memory| memory.borrow().len() as u64);
    if end > memory_len {
        wasm_atomic_trap("atomic-oob");
    }
}

fn wasm_atomic_addr(addr: i32, offset: u64, width: u64) -> i32 {
    wasm_atomic_range(u64::from(addr as u32), offset, width);
    addr
}

fn wasm_atomic_addr_a64(addr: i64, offset: u64, width: u64) -> i64 {
    wasm_atomic_range(addr as u64, offset, width);
    addr
}

fn wasm_load_i32(addr: i32, offset: u64) -> i32 {
    WASM_MEMORY.with(|m| {
        let m = m.borrow();
        let a = wasm_mem_addr(addr, offset);
        let mut b = [0u8; 4];
        if a + 4 <= m.len() { b.copy_from_slice(&m[a..a + 4]); }
        i32::from_le_bytes(b)
    })
}
fn wasm_load_i64(addr: i32, offset: u64) -> i64 {
    WASM_MEMORY.with(|m| {
        let m = m.borrow();
        let a = wasm_mem_addr(addr, offset);
        let mut b = [0u8; 8];
        if a + 8 <= m.len() { b.copy_from_slice(&m[a..a + 8]); }
        i64::from_le_bytes(b)
    })
}
fn wasm_load_f32(addr: i32, offset: u64) -> f32 { f32::from_bits(wasm_load_i32(addr, offset) as u32) }
fn wasm_load_f64(addr: i32, offset: u64) -> f64 { f64::from_bits(wasm_load_i64(addr, offset) as u64) }
fn wasm_load_i32_8u(addr: i32, offset: u64) -> i32 {
    WASM_MEMORY.with(|m| { let m = m.borrow(); let a = wasm_mem_addr(addr, offset); *m.get(a).unwrap_or(&0) as i32 })
}

}

fn wasm_throw_ref(tag: i64) { WASM_EXCEPTION.with(|e| *e.borrow_mut() = Some(tag)); }
fn wasm_exception_pending(tag: Option<i64>) -> bool {
    WASM_EXCEPTION.with(|e| match (*e.borrow(), tag) {
        (Some(t), Some(want)) => t == want,
        (Some(_), None) => true,
        _ => false,
    })
}

fn wasm_memory_size() -> i32 {
    WASM_MEMORY.with(|m| (m.borrow().len() / (64 * 1024)) as i32)
}
fn wasm_memory_grow(delta: i32) -> i32 {
    WASM_MEMORY.with(|m| {
        let mut m = m.borrow_mut();
        let prev = (m.len() / (64 * 1024)) as i32;
        if delta > 0 {
            let new_len = m.len() + (delta as usize) * 64 * 1024;
            m.resize(new_len, 0u8);
        }
        prev
    })
}

thread_local! {
    static WASM_TABLES: RefCell<Vec<Vec<usize>>> = RefCell::new(vec![vec![0usize; 0]; 8]);
}

fn wasm_table_get(table: u32, index: i32) -> usize {
    WASM_TABLES.with(|t| t.borrow().get(table as usize).and_then(|v| v.get(index as usize)).copied().unwrap_or(0))
}
fn wasm_table_set(table: u32, index: i32, value: usize) {
    WASM_TABLES.with(|t| { let mut t = t.borrow_mut(); if let Some(v) = t.get_mut(table as usize) { if (index as usize) < v.len() { v[index as usize] = value; } } });
}
fn wasm_table_size(table: u32) -> i32 {
    WASM_TABLES.with(|t| t.borrow().get(table as usize).map_or(0, |v| v.len() as i32))
}
fn wasm_table_grow(table: u32, init: usize, n: i32) -> i32 {
    WASM_TABLES.with(|t| { let mut t = t.borrow_mut(); if let Some(v) = t.get_mut(table as usize) { let prev = v.len() as i32; if n > 0 { v.extend(std::iter::repeat_n(init, n as usize)); } prev } else { -1 } })
}
fn wasm_table_fill(table: u32, index: i32, value: usize, n: i32) {
    WASM_TABLES.with(|t| { let mut t = t.borrow_mut(); if let Some(v) = t.get_mut(table as usize) { for i in 0..(n.max(0) as usize) { let p = index as usize + i; if p < v.len() { v[p] = value; } } } });
}
fn wasm_table_copy(dst_table: u32, src_table: u32, dst: i32, src: i32, n: i32) {
    let copied: Vec<usize> = WASM_TABLES.with(|t| { let t = t.borrow(); (0..(n.max(0) as usize)).map(|i| t.get(src_table as usize).and_then(|v| v.get(src as usize + i)).copied().unwrap_or(0)).collect() });
    WASM_TABLES.with(|t| { let mut t = t.borrow_mut(); if let Some(v) = t.get_mut(dst_table as usize) { for (i, val) in copied.into_iter().enumerate() { let p = dst as usize + i; if p < v.len() { v[p] = val; } } } });
}
fn wasm_table_init(table: u32, _elem: u32, dst: i32, _src: i32, n: i32) {
    WASM_TABLES.with(|t| { let mut t = t.borrow_mut(); if let Some(v) = t.get_mut(table as usize) { for i in 0..(n.max(0) as usize) { let p = dst as usize + i; if p < v.len() { v[p] = 0; } } } });
}

fn wasm_memory_discard(dst: usize, n: usize) {
    WASM_MEMORY.with(|m| { let mut m = m.borrow_mut(); if dst.saturating_add(n) <= m.len() { for byte in &mut m[dst..dst + n] { *byte = 0u8; } } });
}

fn wasm_global_atomic_rmw(index: u32, val: i64, op: &str) -> i64 {
    WASM_GLOBALS.with(|g| { let mut g = g.borrow_mut(); let slot = g.get_mut(index as usize); match slot { Some(s) => { let prev = *s; *s = match op { "add" => prev.wrapping_add(val), "sub" => prev.wrapping_sub(val), "and" => prev & val, "or" => prev | val, "xor" => prev ^ val, _ => val }; prev } None => 0 } })
}
fn wasm_global_atomic_cmpxchg(index: u32, expected: i64, replacement: i64) -> i64 {
    WASM_GLOBALS.with(|g| { let mut g = g.borrow_mut(); match g.get_mut(index as usize) { Some(s) => { let prev = *s; if prev == expected { *s = replacement; } prev } None => 0 } })
}

fn wasm_i64_add128(a_lo: i64, a_hi: i64, b_lo: i64, b_hi: i64) -> (i64, i64) {
    let a: i128 = ((a_hi as i128) << 64) | (a_lo as u64 as i128);
    let b: i128 = ((b_hi as i128) << 64) | (b_lo as u64 as i128);
    let r: i128 = a.wrapping_add(b);
    (r as i64, (r >> 64) as i64)
}
fn wasm_i64_sub128(a_lo: i64, a_hi: i64, b_lo: i64, b_hi: i64) -> (i64, i64) {
    let a: i128 = ((a_hi as i128) << 64) | (a_lo as u64 as i128);
    let b: i128 = ((b_hi as i128) << 64) | (b_lo as u64 as i128);
    let r: i128 = a.wrapping_sub(b);
    (r as i64, (r >> 64) as i64)
}
fn wasm_i64_mul_wide_s(a: i64, b: i64) -> (i64, i64) {
    let r: i128 = (a as i128) * (b as i128);
    (r as i64, (r >> 64) as i64)
}
fn wasm_i64_mul_wide_u(a: i64, b: i64) -> (i64, i64) {
    let r: u128 = (a as u64 as u128) * (b as u64 as u128);
    (r as i64, (r >> 64) as i64)
}
